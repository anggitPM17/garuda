project(Garuda VERSION 67.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)

  set(CMAKE_BUILD_TYPE Release)

if(MSVC)
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /Brepro")
  set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} /Brepro")
  set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} /Brepro")
  set(CMAKE_MODULE_LINKER_FLAGS "${CMAKE_MODULE_LINKER_FLAGS} /Brepro")

option(GARUDA_BUILD_TESTS "Bangun tes dan benchmark" ON)

option(GARUDA_STATIC_LINK "Link MSVC /MT atau GCC/Clang static runtime" ON)

if(GARUDA_STATIC_LINK)
  if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    foreach(flag_var CMAKE_C_FLAGS CMAKE_C_FLAGS_DEBUG CMAKE_C_FLAGS_RELEASE CMAKE_CXX_FLAGS CMAKE_CXX_FLAGS_DEBUG CMAKE_CXX_FLAGS_RELEASE)
      if(${flag_var} MATCHES "/MD")
        string(REGEX REPLACE "/MD" "/MT" ${flag_var} "${${flag_var}}")
      endif()
    endforeach()
  else()
    set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -static-libgcc -static-libstdc++")
  endif()
endif()

find_package(Threads REQUIRED)

add_library(sqlite3 STATIC src/vendor/sqlite3/sqlite3.c)
target_include_directories(sqlite3 PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src/vendor/sqlite3>
  $<INSTALL_INTERFACE:include>)
set_target_properties(sqlite3 PROPERTIES LINKER_LANGUAGE CXX)
if(MSVC)
  target_compile_options(sqlite3 PRIVATE /W0)
endif()

file(GLOB MBEDTLS_SOURCES "src/vendor/mbedtls/library/*.c")
add_library(mbedtls STATIC ${MBEDTLS_SOURCES})
target_include_directories(mbedtls PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src/vendor/mbedtls/include>
  $<INSTALL_INTERFACE:include>)
set_target_properties(mbedtls PROPERTIES LINKER_LANGUAGE CXX)
if(MSVC)
  target_compile_options(mbedtls PRIVATE /W0)
  target_link_libraries(mbedtls PUBLIC bcrypt)
endif()

  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion ARCH_INDEPENDENT)

"include(CMakeFindDependencyMacro)\nfind_dependency(Threads)\ninclude(\"\${CMAKE_CURRENT_LIST_DIR}/GarudaTargets.cmake\")\n")
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/GarudaConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/GarudaConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Garuda)
